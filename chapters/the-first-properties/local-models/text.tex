\subsection{Local models}

We begin from the regular local models, more explictly, the unit open ball $B$ in $\bbC^n$.
They are local models of complex manifolds.

\begin{definition}\label{def:sheaf-of-holomorphic-functions-on-unit-open-ball}
	Let \(B \subset \bbC^n\) be the unit open ball.
	The \emph{sheaf of holomorphic functions} on \(B\), denoted by \(\scrO_B\),
	is the assignment that to each open subset \(U \subset B\) assigns the ring \(\scrO_B(U)\) of all holomorphic functions $f:U \to \bbC$ on \(U\),
	and set the restriction as the usual restriction of functions.
\end{definition}

In general, \(\scrO_B(U)\) is not a noetherian ring for an open set \(U \subset B\).

\begin{example}\label{eg:ring-of-holomorphic-functions-is-not-noetherian}
	Let $B \subset \bbC$ be the unit open ball.
	Set $I_k = \{f \in \scrO_B(B) | f(1/2) = \cdots = f(1/k) = 0\}$ be ideals of $\scrO_B(B)$.
	Then $f_k = (z-1/2)\cdots (z-1/k) \in I_k \setminus I_{k+1}$ and hence we get a strictly descending chain $I_2 \supseteq I_3 \supseteq \cdots$ of ideals of $\scrO_B(B)$.
\end{example}

A fundamental property of the sheaf of holomorphic functions is its coherence.
By \cref{eg:ring-of-holomorphic-functions-is-not-noetherian}, the coherence is surprising.

	The sheaf of holomorphic functions \(\calO_B\) on the open unit ball $B$ is a coherent sheaf.
\end{theorem}

\begin{definition}\label{def:local-model-of-complex-analytic-spaces}
	A \emph{local model of complex analytic spaces} is a locally ringed space with structure sheaf of functions which is isomorphic to $(Z,\scrO_Z \subset \scrC_{Z,\bbC}^0)$ for some $Z = Z(f_1,\ldots,f_k;B)$.
	A \emph{regular local model of complex analytic spaces} is a locally ringed space with structure sheaf of functions which is isomorphic to an open unit ball $(B,\scrO_B \subset \scrC_{B,\bbC}^0)$.
\end{definition}

Now let us discuss the differential.
First we begin from the nonsingular case.

\begin{construction}
	Let $B$ be a $n$-dimensional regular local model of complex analytic spaces.
	By definition, $(B,\scrC_{B,\bbR}^\infty \subset \scrC_{B,\bbR}^0)$ carries a natural structure of real manifolds.
	Hence we can consider its cotangent bundle $T^*B$.
	Its smooth sections forms a $\scrC_{B,\bbR}^\infty$-module $\scrF_{\sm,\bbR}$, and there is a natural $\underline{\bbR}$-linear map $\upd_{\sm,\bbR}$ sending smooth functions to smooth sections in $T^*B$.
	As $\scrC_{B,\bbR}^\infty$-module, it is locally free of rank $2n$.

	Above is totally of real coefficients.
	On the other hands, we can also consider holomorphic sections in $T^*B$.
	This sheaf $\scrF_{\hol,\bbC}$ is an $\scrO_B$-module.
	And we also have a natural $\underline{\bbC}$-linear map $\upd_{\hol,\bbC}$ sending holomorphic functions to holomorphic sections by definition of holomorphicity.
	As $\scrO_{B}$-module, $\scrF_{\hol,\bbC}$ is locally free of rank $n$.

	We extend both of them to $\scrC_{B,\bbC}^\infty$.
	Set
	\[ \scrA_B^1 \coloneqq \scrF_{\sm,\bbR} \otimes_{\scrC^{\infty}_{B,\bbR}} \scrC^{\infty}_{B,\bbC}, \qquad \scrA_B^{1,0} \coloneqq \scrF_{\hol,\bbC} \otimes_{\scrO_{B}} \scrC^{\infty}_{B,\bbC}. \]
	Since any holomorphic function is naturally smooth as map to $\bbC$, we have an inclusion of sheaves $\scrA_{B}^{1,0} \subset \scrA_B^1$.
	The differential map $\upd_{\sm,\bbR}$ extends to a $\scrC_{B,\bbC}^\infty$-map easily since $\scrA_B^1 = \scrF_{\sm,\bbR} \oplus \scrF_{\sm,\bbR} \cdot \im$ as $\scrC^{\infty}_{B,\bbR}$-modules (this is since we always fix such a $\bbR$-basis of $\bbC$).
	Explicitly, we have a $\underline{\bbC}$-linear map
	\[ \upd_{\sm,\bbC}: \scrC_{B,\bbC}^\infty \to \scrA^1, \qquad \upd_{\sm,\bbC}(u+\im v) = \upd_{\sm,\bbR} u + \im \upd_{\sm,\bbR} v, \quad u,v \in \scrC_{B,\bbR}^\infty(U). \]
	For $f \in \scrO_B(U) \subset \scrC_{B,\bbC}^\infty(U)$, we have $\upd_{\hol,\bbC}f = \upd_{\sm,\bbC}f$ by definition.

	A subtle thing is that there are two $\underline{\bbC}$-module sturcture on $\scrA_B^1$.
	One is given by the smooth $\bbR$-linear map $J: T^*B \to T^*B, v \mapsto \im v$ (note that $T^*B$ is naturally a smooth $\bbC$-vector bundle).
	This is also called \emph{almost complex structure}; see \Yang{}.
	The other comes from the tensor by $\scrC^{\infty}_{B,\bbC}$, which is clearly a $\underline{\bbC}$-algebra.
	We use ths latter as the $\underline{\bbC}$-module sturcture on $\scrA_B^1$ (so that $\upd_{\sm,\bbC}$ is $\underline{\bbC}$-linear).
	Hence the first $J$ will be a $\underline{\bbC}$-linear map comes from a $\underline{\bbR}$-linear map before tensoring.
	The sheaf $\scrA_B^1$ splits into two subsheaves of rank $n$ corresponding to the eigenvalue $\im$ and $-\im$ of $J$.
	On subsheaf associated to $\im$, two $\underline{\bbC}$-module sturctures coincide and on another part, the two $\underline{\bbC}$-module sturctures are conjugation over $\bbR$.

	Note that on $\scrA_B^{1,0}$, there are only one $\underline{\bbC}$-module sturcture.
	In other word, the two $\underline{\bbC}$-module sturctures comes from $\scrA_B^{1}$ coincides and $\scrA_B^{1,0}$ is the subsheaf defined by $J = \im$.
	Set $\scrA_B^{0,1}$ be the subsheaf of $\scrA_B^1$ defined by $J = -\im$.
	Then we have
	\[ \scrA^1_B \coloneqq \scrA_B^{1,0} \oplus \scrA_{B}^{0,1}. \]
	% Also write $\upd_{\sm,\bbC} = \upd^{1,0} + \upd^{0,1}$ according to the decomposition.
\end{construction}

\begin{definition}
	Let $B$ be a regular local model of complex analytic spaces.
	The \emph{sheaf of complex-valued smooth $1$-differential forms} (resp. \emph{sheaf of complex-valued smooth $(1,0)$-differential forms}, \emph{sheaf of complex-valued smooth $(0,1)$-differential forms}) is the sheaf $\scrA_B^1$ (resp. $\scrA_B^{1,0}$, $\scrA_B^{0,1}$ defined in \Yang{}.
	And the \emph{differential map $\upd$} is defined as the map $\upd_{\sm,\bbC}$ in \Yang{}.
	According to the decomposition $\scrA_B^1 = \scrA_B^{1,0} \oplus \scrA_B^{0,1}$, we write $\upd = \partial + \partialbar$.

	The \emph{sheaf of complex-valued smooth $k$-differential forms}, \emph{sheaf of complex-valued smooth $(p,q)$-differential forms} and \emph{sheaf of complex-valued smooth differential forms} are defined as
	\[ \scrA_B^k \coloneqq \bigwedge^k \scrA_B^1, \qquad \scrA_B^{p,q} \coloneqq \bigwedge^p \scrA_B^{1,0} \otimes \bigwedge^q \scrA_B^{0,1}, \qquad \scrA_B \coloneqq \scrA_B^\bullet \coloneqq \bigoplus_{k\geq 0} \scrA_B^k = \bigoplus_{p,q \geq 0} \scrA_B^{p,q}. \]
	Note that $\scrA_B^k = \bigoplus_{p+q=k} \scrA_B^{p,q}$.
	An element $\omega \in \scrA_B^\bullet(U)$ is called \emph{of degree $k$} (resp. \emph{of bidegree $(p,q)$}) if $\omega \in \scrA_B^k(U)$ (resp. $\omega \in \scrA_B^{p,q}(U)$).
	The differentials
	\[ \upd: \scrA_B^k \to \scrA_B^{k+1}, \qquad \partial: \scrA_B^{p,q} \to \scrA_B^{p+1,q}, \qquad \partialbar: \scrA_B^{p,q} \to \scrA_B^{p,q+1} \]
	extend by Leibinz\Yang{} rule.
\end{definition}
